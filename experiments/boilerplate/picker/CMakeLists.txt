cmake_minimum_required(VERSION 3.16)
project(brick_picker LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PICKER_DEPS REQUIRED IMPORTED_TARGET xapian-core json-c libcurl)
add_executable(brick-picker picker.cpp)
target_compile_features(brick-picker PRIVATE cxx_std_17)
target_include_directories(brick-picker PRIVATE ../../../include)
target_link_libraries(brick-picker PRIVATE PkgConfig::PICKER_DEPS)

find_path(UDPIPE_INCLUDE_DIR udpipe.h)
find_library(UDPIPE_LIBRARY NAMES udpipe)
if(UDPIPE_INCLUDE_DIR AND UDPIPE_LIBRARY)
    add_executable(brick-requirements requirements.cpp)
    target_compile_features(brick-requirements PRIVATE cxx_std_17)
    target_include_directories(brick-requirements PRIVATE ../../../include ${UDPIPE_INCLUDE_DIR})
    target_link_libraries(brick-requirements PRIVATE PkgConfig::PICKER_DEPS ${UDPIPE_LIBRARY})
endif()

option(PICKER_BGE "Build the optional BGE example matcher when ONNX Runtime is available" ON)
if(PICKER_BGE)
    find_package(onnxruntime QUIET)
    if(onnxruntime_FOUND)
        target_sources(brick-picker PRIVATE ../../../src/meaning.cpp ../../../src/reader.cpp)
        target_compile_definitions(brick-picker PRIVATE UNLLM_HAVE_ONNX)
        target_link_libraries(brick-picker PRIVATE onnxruntime::onnxruntime)
        set(picker_test_feature bge)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_test(NAME picker_contract COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/test.sh" $<TARGET_FILE:brick-picker> ${picker_test_feature})
    set(PICKER_UDPIPE_TEST_MODEL "" CACHE FILEPATH "English UDPipe model for adapter regression checks")
    if(TARGET brick-requirements AND PICKER_UDPIPE_TEST_MODEL)
        add_test(NAME explicit_requirements COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/test-requirements.sh"
            $<TARGET_FILE:brick-requirements> "${CMAKE_CURRENT_SOURCE_DIR}/requirement-terms.json" "${PICKER_UDPIPE_TEST_MODEL}")
    endif()
endif()
