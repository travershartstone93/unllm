# The pinned SDK includes an unused Java targets file with a pre-3.5 minimum.
# Supply compatibility policy explicitly for CMake 4, without modifying the SDK.
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
find_package(cvc5 1.3.1 EXACT CONFIG REQUIRED PATHS "${UNLLM_ROOT}/.deps/cvc5-native/cvc5-Linux-x86_64-shared/lib/cmake/cvc5" NO_DEFAULT_PATH)
add_library(unllm_native_core
  "${UNLLM_ROOT}/src/symbolic/common.cpp"
  "${UNLLM_ROOT}/src/symbolic/model.cpp"
  "${UNLLM_ROOT}/src/symbolic/store.cpp"
  "${UNLLM_ROOT}/src/symbolic/process.cpp")
find_package(Java REQUIRED COMPONENTS Runtime)
set(ANTLR_JAR "${UNLLM_ROOT}/.deps/apps/archives/antlr-4.13.2-complete.jar")
set(ANTLR_PREFIX "${UNLLM_ROOT}/.deps/apps/native")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
set(PARSER_SOURCES)
foreach(GRAMMAR SimpleMrs AceProtocol Provenance ScalarExpr BehaviourExpr)
  set(OUTPUTS "${GENERATED}/${GRAMMAR}Lexer.cpp" "${GENERATED}/${GRAMMAR}Parser.cpp")
  add_custom_command(OUTPUT ${OUTPUTS} "${GENERATED}/${GRAMMAR}Lexer.h" "${GENERATED}/${GRAMMAR}Parser.h"
    COMMAND "${Java_JAVA_EXECUTABLE}" -jar "${ANTLR_JAR}" -Dlanguage=Cpp -no-listener -o "${GENERATED}" "${GRAMMAR}.g4"
    DEPENDS "${UNLLM_ROOT}/grammars/${GRAMMAR}.g4"
    WORKING_DIRECTORY "${UNLLM_ROOT}/grammars" VERBATIM)
  list(APPEND PARSER_SOURCES ${OUTPUTS})
endforeach()
add_library(unllm_native_parsers ${PARSER_SOURCES})
target_include_directories(unllm_native_parsers PUBLIC "${GENERATED}")
target_include_directories(unllm_native_parsers SYSTEM PUBLIC "${ANTLR_PREFIX}/include/antlr4-runtime")
target_link_libraries(unllm_native_parsers PUBLIC "${ANTLR_PREFIX}/lib/libantlr4-runtime.a")
target_sources(unllm_native_core PRIVATE "${UNLLM_ROOT}/src/symbolic/mrs.cpp")
target_link_libraries(unllm_native_core PUBLIC unllm_native_parsers)
target_include_directories(unllm_native_core PUBLIC "${UNLLM_ROOT}/include")
target_link_libraries(unllm_native_core PUBLIC PkgConfig::JSONC ICU::uc OpenSSL::Crypto Threads::Threads ${SQLITE_TARGET})
target_compile_options(unllm_native_core PRIVATE -Wall -Wextra -Wpedantic)
